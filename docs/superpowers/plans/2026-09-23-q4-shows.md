```ts
/**
 * Groups shows into recurring series using ONLY fields the sheet already carries
 * (name + province + city) — no new sheet column. `recurring` (monthly/annual/...)
 * is cadence metadata, not identity: two different sub-series at the same venue
 * ("Card & Comic Show" vs "Pokemon Show") can share the same recurring value, so
 * grouping on it alone would merge them. Checked against the live dataset
 * 2026-09-23: 42 of 95 unique (name, city) combinations have 2+ dates.
 */
export function seriesKey(show: ShowRecord): string {
  return `${show.name.trim().toLowerCase()}|${show.province}|${show.citySlug}`;
}

export function groupShowsIntoSeries(shows: ShowRecord[]): ShowSeries[] {
  const groups = new Map<string, ShowRecord[]>();
  for (const show of shows) {
    const key = seriesKey(show);
    const list = groups.get(key);
    if (list) list.push(show);
    else groups.set(key, [show]);
  }
  const series: ShowSeries[] = [];
  for (const group of groups.values()) {
    if (group.length < 2) continue; // a single date isn't a series worth its own page
    const sorted = [...group].sort((a, b) => parseLocalDate(a.startDate).getTime() - parseLocalDate(b.startDate).getTime());
    const first = sorted[0]!;
    series.push({
      slug: `${slugify(first.name)}-${first.citySlug}`,
      name: first.name,
      city: first.city,
      citySlug: first.citySlug,
      province: first.province,
      shows: sorted,
    });
  }
  return series;
}

/** First not-yet-passed date in a series, or undefined once every date has happened. */
export function nextInSeries(series: ShowSeries, buildDate: Date): ShowRecord | undefined {
  return series.shows.find((s) => isUpcoming(s, buildDate));
}
```

### 3a. Small, testable, headless-safe change: a `build_summary()` function

This is the only part of `refresh-shows.py` this plan touches — it's pure (no Chrome, no network), and it's what a CI step would actually consume (a machine-readable file instead of parsing the markdown report). It does not attempt to make the scrape itself headless-runnable, because it isn't — see the design note below.

```python
def build_summary(today: str, counts: Counter, gone: list[dict], status: dict[str, str]) -> dict:
    """Machine-readable run outcome, written alongside the CSV/MD for a CI step to
    read without parsing markdown. Counts only -- the payload itself stays the
    review artifact; this is just "did anything happen" for a workflow to act on."""
    return {
        'date': today,
        'new': counts['NEW'],
        'changed': counts['CHANGED'],
        'known': counts['KNOWN'],
        'gone': len(gone),
        'review_identity': counts['REVIEW-IDENTITY'],
        'review_multiday': counts['REVIEW-MULTIDAY'],
        'previously_rejected': counts['PREVIOUSLY-REJECTED'],
        'provinces_with_errors': sorted(p for p, s in status.items() if s.startswith('error')),
        'needs_review': counts['NEW'] + counts['CHANGED'] + counts['REVIEW-IDENTITY'] + counts['REVIEW-MULTIDAY'] > 0,
    }
```

Called from `main()` right after `report.write_text(...)`, written to `OUTDIR / f'{today}-show-refresh-summary.json'`.

### 3b. Why the scrape itself stays out of CI — decision for Nathan

`check_chrome()`'s own comment is unambiguous: TCDB 403s every non-browser HTTP client, and `dev-browser --connect` to a **signed-in** Chrome is the only route that has worked. A GitHub-hosted runner starts a fresh VM every run with no signed-in browser profile — this script cannot pass `check_chrome()` there, full stop. Two real options, both requiring a call only Nathan can make (new infrastructure / new trust boundary):

1. **Self-hosted runner on the machine that already runs AI Chrome.** The workflow below would then work completely unchanged — same script, same `dev-browser --connect`, just triggered by a schedule instead of by hand. Lowest engineering risk; ties a GitHub Action to a Mac that has to be on and signed in every Monday.
2. **Redesign the scrape to not need a signed-in profile** — e.g. real headless Playwright with a residential-ish fingerprint, or a different source than TCDB. Untested here (no live network access in this worktree to verify TCDB's actual block reason — could be a UA/TLS fingerprint check that a plain headless browser clears, or could be closer to real bot-detection that it doesn't). This is a genuine research task, not a "small change," and shouldn't be attempted blind.
